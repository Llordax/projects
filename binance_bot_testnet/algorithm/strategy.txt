//@version=6
strategy("Estrategia TP/SL Escalonado RSI + EMA50 (TP Ampliado)", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100)

/// === Indicadores === ///
ema50 = ta.ema(close, 50)
rsi = ta.rsi(close, 14)

/// === Gestión de Riesgo === ///
tp1_pct = 1.5 / 100
tp2_pct = 3.0 / 100
tp3_pct = 4.5 / 100

sl1_pct = 0.5 / 100
sl2_pct = 1.0 / 100
sl3_pct = 1.5 / 100

/// === Condición de Entrada === ///
rsiCondition = rsi < 50  // ✅ Más frecuente
trendUp = ema50 > ema50[1]
longCondition = rsiCondition and trendUp

/// === Entradas === ///
if (longCondition)
    strategy.entry("Entrada Escalonada", strategy.long)

/// === TP y SL usando precio de entrada === ///
tp1 = strategy.position_avg_price * (1 + tp1_pct)
tp2 = strategy.position_avg_price * (1 + tp2_pct)
tp3 = strategy.position_avg_price * (1 + tp3_pct)

sl1 = strategy.position_avg_price * (1 - sl1_pct)
sl2 = strategy.position_avg_price * (1 - sl2_pct)
sl3 = strategy.position_avg_price * (1 - sl3_pct)

strategy.exit("TP1", from_entry="Entrada Escalonada", limit=tp1, qty_percent=33)
strategy.exit("TP2", from_entry="Entrada Escalonada", limit=tp2, qty_percent=33)
strategy.exit("TP3", from_entry="Entrada Escalonada", limit=tp3, qty_percent=34)

strategy.exit("SL1", from_entry="Entrada Escalonada", stop=sl1, qty_percent=33)
strategy.exit("SL2", from_entry="Entrada Escalonada", stop=sl2, qty_percent=33)
strategy.exit("SL3", from_entry="Entrada Escalonada", stop=sl3, qty_percent=34)

/// === Visualización === ///
plot(ema50, title="EMA 50", color=color.orange)
hline(50, "RSI 50", color=color.gray)
plot(rsi, title="RSI", color=color.blue)

plotshape(longCondition, title="Señal de Compra", location=location.belowbar, color=color.green, style=shape.labelup, text="LONG")
